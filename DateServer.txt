import java.io.*;
import java.net.*;
import java.util.concurrent.ConcurrentHashMap;

public class DateServer {

    private static final int PORT = 9090;

    private static ConcurrentHashMap<String, ClientInfo> clients = new ConcurrentHashMap<>();

    public static void main(String[] args) throws Exception {

        ServerSocket serverSocket = new ServerSocket(PORT);

        System.out.println("Server running at IP: " + InetAddress.getLocalHost().getHostAddress());
        System.out.println("Listening on port: " + PORT);

    
        new Thread(new ServerCommandHandler()).start();

        while (true) {
            Socket socket = serverSocket.accept();
            ClientHandler handler = new ClientHandler(socket);
            new Thread(handler).start();
        }
    }

    private static class ClientHandler implements Runnable {

        private Socket socket;
        private BufferedReader in;
        private PrintWriter out;
        private String name;

        public ClientHandler(Socket socket) {
            this.socket = socket;
        }

        public void run() {

            try {

                in = new BufferedReader(new InputStreamReader(socket.getInputStream()));
                out = new PrintWriter(socket.getOutputStream(), true);

                while (true) {

                    out.println("ENTER_NAME");
                    name = in.readLine();

                    if (name == null) return;

                    if (!clients.containsKey(name)) break;

                    out.println("NAME_TAKEN");
                }

                ClientInfo info = new ClientInfo(name, socket.getInetAddress().toString(), Thread.currentThread(), out);
                clients.put(name, info);

                System.out.println(name + " connected.");

                broadcast("SERVER: " + name + " joined.");
                sendClientList();

                String message;

                while ((message = in.readLine()) != null) {

                    if (message.equalsIgnoreCase("/list")) {
                        sendClientListTo(out);
                    }

                    else if (message.startsWith("/msg")) {

                        String[] parts = message.split(" ", 3);

                        if (parts.length >= 3)
                            sendPrivate(parts[1], name + ": " + parts[2]);
                    }

                    else if (message.startsWith("/broadcast")) {

                        String text = message.substring(11);
                        broadcast(name + ": " + text);
                    }

                    else if (message.equalsIgnoreCase("/exit")) {
                        break;
                    }
                }

            } catch (Exception e) {
                System.out.println("Connection error.");
            }

            finally {
                disconnect();
            }
        }

        private void disconnect() {

            try {

                if (name != null) {

                    clients.remove(name);
                    broadcast("SERVER: " + name + " disconnected.");
                    sendClientList();
                }

                socket.close();

            } catch (Exception ignored) {}
        }
    }


    private static class ServerCommandHandler implements Runnable {

        public void run() {

            BufferedReader console = new BufferedReader(new InputStreamReader(System.in));

            try {

                String command;

                while ((command = console.readLine()) != null) {

                    if (command.startsWith("/broadcast")) {

                        String text = command.substring(11);
                        broadcast("SERVER: " + text);
                    }

                    else if (command.startsWith("/msg")) {

                        String[] parts = command.split(" ", 3);

                        if (parts.length >= 3)
                            sendPrivate(parts[1], "SERVER (private): " + parts[2]);
                    }

                    else if (command.equals("/list")) {

                        System.out.println("\nConnected Clients:");

                        for (ClientInfo c : clients.values())
                            System.out.println(c.name + " | " + c.ip);

                        System.out.println();
                    }

                    else if (command.startsWith("/kick")) {

                        String name = command.split(" ")[1];
                        forceClose(name);
                    }
                }

            } catch (Exception e) {
                e.printStackTrace();
            }
        }
    }

    private static class ClientInfo {

        String name;
        String ip;
        Thread thread;
        PrintWriter out;

        ClientInfo(String name, String ip, Thread thread, PrintWriter out) {
            this.name = name;
            this.ip = ip;
            this.thread = thread;
            this.out = out;
        }
    }

    private static void broadcast(String msg) {

        for (ClientInfo c : clients.values())
            c.out.println(msg);

        System.out.println(msg);
    }

    private static void sendPrivate(String target, String msg) {

        ClientInfo c = clients.get(target);

        if (c != null)
            c.out.println(msg);
    }

    private static void forceClose(String name) {

    ClientInfo c = clients.get(name);

    if (c != null) {

        c.out.println("SERVER_KICK");

        try {
            c.thread.interrupt();
        } catch (Exception ignored) {}

        clients.remove(name);

        broadcast("SERVER: " + name + " was kicked.");
        sendClientList();
    }
}

    private static void sendClientList() {

        StringBuilder list = new StringBuilder("CLIENT_LIST ");

        for (String name : clients.keySet())
            list.append(name).append(",");

        for (ClientInfo c : clients.values())
            c.out.println(list.toString());
    }

    private static void sendClientListTo(PrintWriter out) {

        out.println("Connected Clients:");

        for (ClientInfo c : clients.values())
            out.println(c.name + " | " + c.ip);

        out.println("END_LIST");
    }
}
